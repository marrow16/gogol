Shortcuts can execute javascript using the following action:
javascript:filename
• 
where filename is either an absolute path - or is taken as a relative path from [user]/Documents/GoGoL/
• 
the script is compiled as ECMAScript 5.1 - any compilation errors are logged to [user]/Documents/GoGoL/errors.log
• 
the script is not cached - it is compiled from the file each time the shortcut is run. Therefore, between shortcut runs, the script can be modified with external editors.

prompt(string[,string...])Prompts the user for input. The prompt is displayed in the statusbar and the function is blocking until input is acquired.
The first string arg is the label for the prompt. Successive string args are allowed responses (case insensitive):
• 
if allowed responses are specified, the function returns as soon as the user types one of them (e.g. key input).
•

promptDialog(string,array[,callback])Prompts the user to complete a dialog. The dialog is displayed from the statusbar and the function is blocking until the user completes the dialog (by pressing either the Ok or Cancel buttons).
The first string arg is the label shown in the statusbar for the dialog. The second array arg is the definition for the dialog.
Returns: object - the dialog response values (or undefined if the user presses the Cancel button)

Example - Save AnimationThe following javascript demonstrates adding a UI dialog to save the record instrument animation - allowing the user to define the start frame, maximum frames and optional metadata (which the standard UI does not support).
// turn exceptions on...
exceptions = true;
// this function is used by the dialog onChange...
function onChange(res, dlg) {
  switch (res["name"]) {
    case "file":
      dlg.Hide("err"); // clear any displayed error when file input is changed
      // update the format radio buttons according to file extension...
      const path = res["value"];
      const ext = path.slice(path.lastIndexOf('.'));
      const curr = dlg.Get("format");
      if (ext === ".gif" && curr !== "Gif") {
        dlg.Set("format", "Gif");
      }
      if (ext === ".mp4" && curr != "Mp4") {
        dlg.Set("format", "Mp4");
      }
      break;
    case "start":
      // make sure start+max frames does not exceed available frames...
      available = dlg.Get("available");
      start = res["value"];
      frames = dlg.Get("frames");
      if ((start + frames) > available) {
        dlg.Set("frames", available - start);
      }
      break;
    case "frames":
      // make sure start+max frames does not exceed available frames...
      available = dlg.Get("available");
      start = dlg.Get("start");
      frames = res["value"];
      if ((start + frames) > available) {
        dlg.Set("frames", available - start);
      }
      break;
  }
}
// dynamic dialog controls...

let ri = recordInstrument();
if (ri) {
  if (ri.FramesCount() > 0) {
    let available = ri.FramesCount();
    // setup dialog controls according to record instrument...
    framesAvailable["label"] = "Frames available: "+available;
    framesStart["max"] = available-1;
    framesMax["max"] = available;
    // show the dialog and wait for response...
    const response = promptDialog("Save Animation", dialog);
    if (response) {
      // a response indicates the dialog wasn't cancelled
      status("Saving...");
      // save the overridden settings...
      const wasFmt = animationFormat();
      const wasCellSize = cellSize();
      const wasBorders = borders();
      try {
        if (response["format"] === "Mp4") {
          animationFormat("mp4");
        } else {
          animationFormat("gif");
        }
        borders(response["borders"]);
        cellSize(response["cellSize"]);
        let metadata = null;
        if (response["useMetadata"]) {
          const str = response["metadata"];
          metadata = Object.fromEntries(
            str.trim().split("\n").flatMap(line => {
              const i = line.indexOf(":");
              return i < 0 ? [] : [[line.slice(0, i).trim(), line.slice(i + 1).trim()]];
            })
          );
        }
        animationSave(response["file"], response["start"], response["frames"], metadata);
        status("Saved");
      } catch (e) {
        status("Save error (see log)");
      } finally {
        // restore overridden settings...
        animationFormat(wasFmt);
        cellSize(wasCellSize);
        borders(wasBorders);
      }
    }
  } else {
    status("No recorded steps");
  }
} else {
  status("Record not enabled");
}
